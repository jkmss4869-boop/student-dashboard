function doGet(e) {
  var studentName = e.parameter.student; 
  var format = e.parameter.format;
  
  if (!studentName) {
    return HtmlService.createHtmlOutput("<h3 style='text-align:center; font-family:sans-serif; color: #fff;'>Vui lòng nhập tên học sinh vào link!</h3>");
  }

  // Kết nối thẳng vào file Sheet hiện tại đang chứa code
  var ss = SpreadsheetApp.openById("1PprVNjzWKQRN-u2Ix_aLQjlYyXsPgpJs4QniLNL690E");
  var sheet = ss.getSheetByName(studentName);

  if (!sheet) {
    return HtmlService.createHtmlOutput("<h3 style='text-align:center; font-family:sans-serif; color: #fff;'>Không tìm thấy dữ liệu của học sinh này.</h3>");
  }

  // Lấy dữ liệu tổng quan ở hàng số 2
  var summaryData = sheet.getRange("B2:D2").getValues()[0];
  var totalCount = summaryData[0] || 0;     // Ô B2: Tổng số bài
  var avgScore = summaryData[1] || 0;       // Ô C2: Điểm trung bình
  var completedCount = summaryData[2] || 0; // Ô D2: Bài hoàn thành

  // ============================================================
  // BLOCK JSON CHO TRANG DASHBOARD
  // ============================================================
  if (format === 'json') {
    var jsonData = JSON.stringify({
      total: totalCount,
      completed: completedCount,
      average: (Math.round(avgScore * 10) / 10).toFixed(1), // Làm tròn 1 chữ số thập phân (vd: 9.2)
      latestImage: "https://ui-avatars.com/api/?name=" + encodeURIComponent(studentName) + "&background=1abc9c&color=fff&size=800"
    });

    var output = ContentService.createTextOutput(jsonData);
    output.setMimeType(ContentService.MimeType.JSON);
    return output;
  }

  // ============================================================
  // BLOCK HTML GIAO DIỆN HIỂN THỊ DANH SÁCH BÀI TẬP (DARK MODE)
  // ============================================================
  
  // Lấy danh sách bài tập từ dòng 3 trở xuống (B, C, D, E)
  var lastRow = sheet.getLastRow();
  var data = (lastRow >= 3) ? sheet.getRange("B3:E" + lastRow).getValues() : [];

  var htmlOutput = `
    <style>
      body { 
        font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif; 
        margin: 0; 
        padding: 20px; 
        background: #121212; 
      }
      .container { 
        display: flex; 
        flex-wrap: wrap; 
        gap: 20px; 
        justify-content: center;
      }
      .card-link {
        text-decoration: none;
        color: inherit;
        display: block;
      }
      .card { 
        width: 280px; 
        border-radius: 16px; 
        box-shadow: 0 8px 24px rgba(0,0,0,0.4); 
        background: #1e1e1e; 
        overflow: hidden; 
        border: 1px solid #2a2a2a;
        transition: transform 0.3s ease, border-color 0.3s ease, box-shadow 0.3s ease;
      }
      .card:hover { 
        transform: translateY(-6px); 
        border-color: #3b82f6;
        box-shadow: 0 12px 28px rgba(59, 130, 246, 0.2); 
      }
      /* Khu vực cover tự động tạo màu gradient */
      .cover { 
        width: 100%; 
        height: 140px; 
        display: flex; 
        align-items: center; 
        justify-content: center; 
        color: white; 
        font-size: 20px; 
        font-weight: 800; 
        text-align: center; 
        padding: 15px; 
        box-sizing: border-box;
        text-shadow: 1px 1px 4px rgba(0,0,0,0.3);
        line-height: 1.3;
      }
      .card-body { 
        padding: 20px; 
        text-align: center; 
      }
      .title { 
        font-size: 15px; 
        font-weight: 700; 
        color: #f3f4f6; 
        margin-bottom: 8px; 
        text-transform: uppercase;
        white-space: nowrap;
        overflow: hidden;
        text-overflow: ellipsis;
      }
      .score {
        font-size: 14px;
        color: #9ca3af;
        margin-bottom: 15px;
      }
      .score strong { color: #f87171; font-size: 17px; }
      
      /* Tag trạng thái */
      .status { 
        display: inline-block; 
        padding: 6px 16px; 
        border-radius: 50px; 
        font-size: 13px; 
        font-weight: 600;
      }
      .status.done { background: rgba(26, 188, 156, 0.15); color: #2ecc71; }
      .status.pending { background: rgba(230, 126, 34, 0.15); color: #fb923c; }
    </style>
    <div class="container">
  `;
  
  // Bộ sưu tập màu sắc Gradient hiện đại và dịu mắt trên nền tối
  var gradients = [
    "linear-gradient(135deg, #4f46e5 0%, #7c3aed 100%)",
    "linear-gradient(135deg, #0ea5e9 0%, #2563eb 100%)",
    "linear-gradient(135deg, #d97706 0%, #dc2626 100%)",
    "linear-gradient(135deg, #059669 0%, #0d9488 100%)",
    "linear-gradient(135deg, #db2777 0%, #9333ea 100%)"
  ];

  for (var i = 0; i < data.length; i++) {
    var lessonName = data[i][0];
    
    if (lessonName !== "") { 
      var score = data[i][1];
      var status = data[i][2];
      var lessonLink = data[i][3] || "#"; // Cột E (index 3) chứa link bài học
      
      // Tự động chọn 1 màu gradient theo thứ tự
      var colorIndex = i % gradients.length;
      var bg = gradients[colorIndex];
      
      var statusClass = (status === "Hoàn thành") ? "done" : "pending";
      var displayScore = (score !== "") ? (Math.round(score * 10) / 10) : "Đang chờ";

      htmlOutput += `
        <a href="${lessonLink}" target="_blank" class="card-link">
          <div class="card">
            <div class="cover" style="background: ${bg}">
              ${lessonName}
            </div>
            <div class="card-body">
              <div class="title" title="${lessonName}">${lessonName}</div>
              <div class="score">Điểm: <strong>${displayScore}</strong></div>
              <div class="status ${statusClass}">${status}</div>
            </div>
          </div>
        </a>
      `;
    }
  }
  
  htmlOutput += "</div>";
  return HtmlService.createHtmlOutput(htmlOutput);
}
