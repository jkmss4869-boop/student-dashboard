    <script>
      // Lấy unit_id từ backend
      var unitId = "<?= currentUnitId ?>";
      
      // Cập nhật tiêu đề (có thể tùy chỉnh thêm để lấy tên Unit đẹp hơn)
      document.getElementById('unit-title').innerText = unitId.toUpperCase();

      // Gọi hàm lấy dữ liệu từ Apps Script
      google.script.run.withSuccessHandler(renderButtons).getExercisesForUnit(unitId);

      function renderButtons(exercises) {
        var container = document.getElementById('button-container');
        container.innerHTML = ''; // Xóa thông báo đang tải

        if (exercises.length === 0) {
          container.innerHTML = "<p>Chưa có bài tập nào cho Unit này.</p>";
          return;
        }

        exercises.forEach(function(ex) {
          var btn = document.createElement('a'); // Dùng thẻ <a> để dễ chuyển hướng
          btn.className = 'exercise-button';
          btn.innerText = ex.title;
          btn.href = ex.link; 
          // Mở trong tab hiện tại (để học sinh vẫn ở trong không gian học)
          // Nếu muốn mở tab mới, dùng: btn.target = "_blank";
          
          container.appendChild(btn);
        });
      }
    </script>
